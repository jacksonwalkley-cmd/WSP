function suggestWeight(name, lastWeight, lastReps) {
  if (!lastWeight) return {target:'4×5', weight:'', reps:''};
  const w = parseInt(lastWeight); const r = parseInt(lastReps);
  if (isNaN(w)) return {target:'4×5', weight:lastWeight, reps:lastReps};
  const isCompound = ['Back Squat','Bench Press','Deadlift','Front Squat','Overhead Press'].includes(name);
  const increment = isCompound ? 5 : 2.5;
  if (r >= 5) return {target:'4×5', weight: w + increment, reps: 5};
  return {target:'4×5', weight: w, reps: lastReps};
}

}

  document.getElementById('prSquat').textContent = '315';
  document.getElementById('prSnap').textContent = '0.72s';
  document.getElementById('prStreak').textContent = state.streak;
  const squatData = [265, 275, 285, 295, 160];
  const squatLabels = ['W1','W2','W3','W4','Del'];
  const maxSquat = Math.max(...squatData);
  document.getElementById('squatChart').innerHTML = squatData.map((v,i) => {
    const h = (v / maxSquat) * 100; const op = i === 4 ? '0.35' : '0.8';
    return '<div class="chart-bar" style="height:'+h+'%;opacity:'+op+'"><span class="val">'+v+'</span><span class="label">'+squatLabels[i]+'</span></div>';
  }).join('');
  const spiralData = [40, 55, 60, 75, 85, 90];
  const spiralLabels = ['S1','S2','S3','S4','S5','S6'];
  document.getElementById('spiralChart').innerHTML = spiralData.map((v,i) => {
    const h = v;
    return '<div class="chart-bar" style="height:'+h+'%"><span class="val">'+v+'%</span><span class="label">'+spiralLabels[i]+'</span></div>';
  }).join('');
  const sprintData = [2.45, 2.38, 2.31, 2.24];
  const sprintLabels = ['W1','W2','W3','W4'];
  const maxSprint = Math.max(...sprintData);
  document.getElementById('sprintChart').innerHTML = sprintData.map((v,i) => {
    const h = (v / maxSprint) * 100;
    return '<div class="chart-bar" style="height:'+h+'%"><span class="val">'+v+'</span><span class="label">'+sprintLabels[i]+'</span></div>';
  }).join('');
}

function renderChat() {
  const ch = document.getElementById('chatHistory');
  ch.innerHTML = state.chat.map(m => '<div class="chat-bubble '+m.sender+'">'+m.text+'</div>').join('');
  ch.scrollTop = ch.scrollHeight;
}

function sendChat() {
  const input = document.getElementById('chatInput');
  const text = input.value.trim(); if (!text) return;
  state.chat.push({sender:'user', text: text});
  input.value = '';
  saveState(); renderChat();
  setTimeout(() => {
    const reply = generateCoachReply(text);
    state.chat.push({sender:'ai', text: reply});
    saveState(); renderChat();
  }, 800);
}

function generateCoachReply(userText) {
  const lower = userText.toLowerCase();
  const firstName = state.profile.name.split(' ')[0];
  if (lower.includes('weight') || lower.includes('squat') || lower.includes('lift')) {
    const lastSquat = state.liftLogs['Wednesday_Back Squat'] || {weight:'285', reps:'5'};
    return firstName + ', your Back Squat is at '+lastSquat.weight+' lbs for '+lastSquat.reps+' reps. If you hit all sets next session, we will bump to '+(parseInt(lastSquat.weight)+5)+' lbs. Trust the progression.';
  }
  if (lower.includes('spiral') || lower.includes('wobble') || lower.includes('tight')) {
    const recent = state.logs.slice(-5);
    const tight = recent.filter(l => l.spiral === 'Tight').length;
    if (tight >= 3) return 'Your spiral consistency is trending up — '+tight+' of your last 5 reps at 14–15 YD were Tight. The follow-through work is paying off.';
    return 'I see some inconsistency in your recent 14–15 YD reps. Let us revisit the Follow-Through Extension drill before your next session. Film one rep slow-motion so we can check hip rotation.';
  }
  if (lower.includes('drill') || lower.includes('fix') || lower.includes('help')) {
    if (state.profile.flaws.length > 0) {
      return 'Based on your profile flaws ('+state.profile.flaws.join(', ')+'), I recommend starting with the Hip Rotation Drill and Follow-Through Extension. Do 2 slow reps of each before every snap day.';
    }
    return 'Let us diagnose. Film a 14–15 YD rep and tell me what feels off — I will match it to a drill.';
  }
  if (lower.includes('recruit') || lower.includes('camp') || lower.includes('coach') || lower.includes('film')) {
    return 'For recruiting, your film should lead with 3 consecutive tight spirals at 14–15 YD under 0.75s. Check the Learn section for "Making Film Clips That Get Watched." Coaches want to see consistency first, speed second.';
  }
  if (lower.includes('streak') || lower.includes('motivation')) {
    return 'You are on a '+state.streak+'-day streak. That discipline compounds — not just physically, but mentally. Show up again tomorrow. The work is the reward.';
  }
  return 'Got it, '+firstName+'. I am tracking that. Keep logging your sessions and I will spot patterns across your block. Consistency over intensity — that is how you get recruited.';
}
